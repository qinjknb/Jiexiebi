def crc16(data, crc=0xFFFF):
    for x in data:
        crc ^= (x << 8) & 0xFFFF
        for _ in range(8):
            crc = ((crc << 1) ^ 0x1021) & 0xFFFF if crc & 0x8000 else (crc << 1) & 0xFFFF
    return crc


def frame(cmd, payload=b"", seq=1):
    body = bytes([cmd, seq, len(payload)]) + payload
    c = crc16(body)
    return b"\xaa\x55" + body + bytes([c >> 8, c & 0xFF])


def enc(ticks):
    out = bytearray()
